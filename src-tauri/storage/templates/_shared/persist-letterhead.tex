% Persistas & Partners — firm letterhead.
%
% This is the firm's identity as it appears on every piece of correspondence
% that leaves the office: the mark, the two partners with their enrolment
% details, and the registered office in the footer. It is separated from
% persist-base.tex because a template may want the firm's typography without
% its letterhead — an internal memo, a draft for comment.
%
% \input{_shared/persist-base} first; this builds on its colours and fonts.
%
% The header and footer are set once here rather than per template. When the
% firm moves office or a partner's number changes, this file changes and every
% document produced afterwards is correct.

\usepackage{tabularx}
\usepackage{changepage}   % adjustwidth, for indented section bodies
\usepackage{ragged2e}     % justifying

% ── Partner block ────────────────────────────────────────────────────────────
% One partner's four lines: name, designation, phone, email, right-aligned.
%
% The leading is set once for the block rather than with \\[1pt] between lines.
% Measured off the firm's own notice: four lines at roughly 10pt apart. Adding
% per-line spacing on top of the font's own leading made the block half again
% as tall and pushed the mark out of line with it.
%
%   \partnerblock{name}{designation}{phone}{email}
\newcommand{\partnerblock}[4]{%
  \begin{minipage}[t]{\linewidth}
    \raggedleft
    \sffamily\fontsize{7.6}{10}\selectfont
    \textbf{#1}\\
    #2\\
    #3\\
    #4
  \end{minipage}%
}

% ── The letterhead itself ────────────────────────────────────────────────────
% Applied to every page, because a notice is served page by page and a page
% without the firm's mark is a page whose provenance is arguable.
%
% A template fills these in with \renewcommand and then calls \applyletterhead.
% Passing thirteen arguments to one macro would be unreadable at the call site
% and unreachable from fancyhdr, which rebuilds the bands on every page.
\newcommand{\firmlogo}{_shared/assets/persistas-logo.png}
\newcommand{\partneronename}{}   \newcommand{\partneronerole}{}
\newcommand{\partneronephone}{}  \newcommand{\partneroneemail}{}
\newcommand{\partnertwoname}{}   \newcommand{\partnertworole}{}
\newcommand{\partnertwophone}{}  \newcommand{\partnertwoemail}{}
\newcommand{\firmwebsite}{}      \newcommand{\firmcontact}{}
\newcommand{\firmofficeone}{}    \newcommand{\firmofficetwo}{}

% The two bands, as content. Defined separately from \applyletterhead so the
% same markup can be a running head on every page or a block set once at the top
% of page one, without being written twice.
\newcommand{\persistheadbands}{%
  \fancyhead[L]{%
    \persistheadtest
    \ifpersistshowhead
      \begin{minipage}[c][\headheight][c]{54mm}
        \includegraphics[width=50mm]{\firmlogo}
      \end{minipage}%
    \fi
  }%
  \fancyhead[R]{%
    \persistheadtest
    \ifpersistshowhead
      \begin{minipage}[c][\headheight][c]{62mm}
        \partnerblock{\partneronename}{\partneronerole}{\partneronephone}{\partneroneemail}\\[5pt]
        % A short rule, right-aligned under the first partner — as on the firm's
        % own notices. A full-width one reads as a section divider.
        \hfill{\color{textPrimary}\rule{46mm}{0.4pt}}\\[6pt]
        \partnerblock{\partnertwoname}{\partnertworole}{\partnertwophone}{\partnertwoemail}
      \end{minipage}%
    \fi
  }%
  \fancyfoot[C]{%
    \persistheadtest
    \begin{minipage}{\textwidth}
      \centering
      \ifpersistshowhead
        {\sffamily\fontsize{7.5}{10}\selectfont
          Website : \textbf{\firmwebsite}\\
          Contact Us: \textbf{\firmcontact}\\
          Regd. Office: \textbf{\firmofficeone}\\
          \textbf{\firmofficetwo}}\\[6pt]
      \fi
      {\fontsize{9}{11}\selectfont\persistpagelabel}
    \end{minipage}%
  }%
}

% The letterhead as a block at the top of the body, for `first` mode.
\newcommand{\persistletterheadblock}{%
  \noindent
  \begin{minipage}[c]{54mm}\includegraphics[width=50mm]{\firmlogo}\end{minipage}%
  \hfill
  \begin{minipage}[c]{62mm}
    \partnerblock{\partneronename}{\partneronerole}{\partneronephone}{\partneroneemail}\\[5pt]
    \hfill{\color{textPrimary}\rule{46mm}{0.4pt}}\\[6pt]
    \partnerblock{\partnertwoname}{\partnertworole}{\partnertwophone}{\partnertwoemail}
  \end{minipage}%
  \par\vspace{8mm}
}

% Compared with \ifx, so it must be a macro rather than a string test.
\def\persistlhfirst{first}

\newcommand{\applyletterhead}{%
  \pagestyle{fancy}%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
  %
  % THE HEAD BAND IS THE WHOLE PROBLEM WITH `first` MODE.
  %
  % \headheight is one value for the document; it cannot be 108pt on page one
  % and 14pt after. Running the letterhead as a head in `first` mode would give
  % every continuation sheet a blank band the height of the firm's mark — a
  % four-centimetre hole at the top of page two.
  %
  % So `first` mode does not use a running head at all. The letterhead is set as
  % body content at the start of the document, the band shrinks to what the page
  % number needs, and page two has an ordinary top margin. Every other mode
  % keeps the running head, because the mark genuinely has to repeat.
  \ifx\persistletterheadmode\persistlhfirst
    \setlength{\headheight}{14pt}%
    \setlength{\footskip}{28pt}%
    \geometry{top=\persistheadmargintop, bottom=\persistheadmarginbottom,
              left=\persistmarginleft, right=\persistmarginright,
              includehead, includefoot,
              headheight=14pt, headsep=6mm, footskip=28pt}%
    \fancyfoot[C]{{\fontsize{9}{11}\selectfont\persistpagelabel}}%
    \AtBeginDocument{\persistletterheadblock}%
  \else
    % Two partner blocks stacked, each four lines, plus the rule between them.
    % Measured rather than guessed: too small and the second partner's email
    % prints over the body of the notice.
    \setlength{\headheight}{108pt}%
    \setlength{\headsep}{14pt}%
    \setlength{\footskip}{56pt}%
    % The page must make room for the band the header now occupies. geometry is
    % re-run rather than set in persist-base, because a template using the base
    % without the letterhead should keep the tighter margins.
    %
    % \geometry, not \newgeometry. \newgeometry re-derives the paper size from
    % the *class* options and discards the papersize persist-base set, so a
    % notice selected as Legal came out A4 — clean compile, right margins, wrong
    % sheet. \applyletterhead runs in the preamble, where \geometry is allowed
    % and accumulates onto what is already set.
    % `includehead`/`includefoot` make `top` and `bottom` the distance from the
    % paper edge to the *bands*, not to the body. Without them the header is
    % pinned to the paper edge and the letterhead sits higher than the firm's own.
    %
    % The margins are the ones the attorney chose — \persistmarginleft and the
    % rest, from persist-layout.tex. This second \geometry overrides the first,
    % so hardcoding them here means a page-setup margin applies to an invoice
    % and is thrown away on the notice it was set for, which is the document
    % that has a filing requirement in the first place.
    \geometry{top=\persistheadmargintop, bottom=\persistheadmarginbottom,
              left=\persistmarginleft, right=\persistmarginright,
              includehead, includefoot,
              headheight=108pt, headsep=10mm, footskip=56pt}%
    \persistheadbands
  \fi
}
% ── Correspondence idioms ────────────────────────────────────────────────────
% Centred, underlined, bold — the firm's house style for the notice type, the
% mode of service and the without-prejudice marker.
\newcommand{\noticebanner}[1]{%
  \begin{center}\underline{\textbf{#1}}\end{center}
}

% "14\textsuperscript{th} July 2026", right-aligned. Indian correspondence
% carries the ordinal; a bare 14/07/2026 reads as a form, not a letter.
\newcommand{\noticedate}[1]{%
  \begin{flushright}\textbf{Date:} #1\end{flushright}
}

% The subject line: bold, justified, and set in caps by the caller — a subject
% is the one part of a notice a recipient always reads.
\newcommand{\noticesubject}[1]{%
  \noindent\textbf{SUBJECT: #1}\par\vspace{2mm}
}

% A numbered section heading, bold, with the number hanging in the margin.
\newcommand{\noticesection}[2]{%
  \vspace{3mm}
  \noindent
  \makebox[0.6cm][l]{\textbf{#1.}}%
  \begin{minipage}[t]{\dimexpr\linewidth-0.6cm\relax}
    \textbf{#2}
  \end{minipage}\par
  \vspace{2mm}
}

% Body paragraphs inside a numbered section sit indented under the heading.
\newenvironment{noticebody}%
  {\begin{adjustwidth}{0.6cm}{0cm}\justifying}%
  {\end{adjustwidth}}

% ── The schedule of payments ─────────────────────────────────────────────────
% A demand notice that says a sum was paid has to say when, how much and how.
% Set as a table rather than as prose: seven tranches in a paragraph is a
% paragraph nobody reads, and a recipient disputing one of them needs to be able
% to point at the row.
%
% The total is not typed. It is summed from these rows by Keel, so the figure
% demanded and the payments said to make it up cannot disagree.
%
%   \begin{paymentschedule}
%     \paymentrow{1}{14 July 2026}{7,000.00}{UPI}{UTR 41822...}
%   \end{paymentschedule}
\newenvironment{paymentschedule}%
  {\vspace{2mm}\par\noindent
   \begin{adjustwidth}{0.6cm}{0cm}
   \begin{tabular}{@{}L{0.8cm}L{3.5cm}R{2.5cm}L{3.0cm}L{3.4cm}@{}}
     \textbf{\footnotesize No.} & \textbf{\footnotesize Date} &
     \textbf{\footnotesize Amount (₹)} & \textbf{\footnotesize Mode} &
     \textbf{\footnotesize Reference} \\[2pt]
     \hline\rule{0pt}{3ex}}%
  {\end{tabular}\end{adjustwidth}\vspace{3mm}}

\newcommand{\paymentrow}[5]{#1. & #2 & #3 & #4 & #5 \\[2pt]}

% The demanded total, set apart from the table so it reads as the demand and not
% as another row.
%   \paymenttotal{1,04,000.00}
\newcommand{\paymenttotal}[1]{%
  \par\noindent
  \begin{adjustwidth}{0.6cm}{0cm}
    \textbf{Total paid: ₹#1/-}
  \end{adjustwidth}
  \vspace{2mm}
}
