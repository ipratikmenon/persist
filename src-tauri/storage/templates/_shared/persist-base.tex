% Persist — shared document preamble.
%
% Every template begins with \input{_shared/persist-base}. The PRD lists
% nineteen templates for the Smart Form Compiler; each carrying its own copy of
% the font, colour and geometry setup would mean nineteen places to change when
% the firm's letterhead does, and nineteen chances to miss one.
%
% ENGINE: XeLaTeX. Not pdfLaTeX — see src-tauri/KEEL-RULES.md.
%
% Templates get: A4 geometry, Noto (with Devanagari declared), the Persist
% colour tokens, the L/C/R column types, and \firmfooter. They add only their
% own content.

% ── Layout ───────────────────────────────────────────────────────────────────
% Paper, margins, typeface, size, spacing, page numbering and which pages carry
% the letterhead are per-document choices, not template content. The drafting
% command compiles them to persist-layout.tex and stages it beside the template;
% everything below reads it. A compile that does not supply one — a test, an
% invoice generated straight from billing — gets these defaults and looks
% exactly as it did before any of this existed.
\InputIfFileExists{persist-layout.tex}{}{}

\providecommand{\persistpaperwidth}{210mm}
\providecommand{\persistpaperheight}{297mm}
\providecommand{\persistmainfont}{Noto Serif}
\providecommand{\persistfontsize}{12.0pt}
\providecommand{\persistleading}{14.4pt}
\providecommand{\persiststretch}{1.00}
\providecommand{\persistbodystyle}{}
\providecommand{\persistmargintop}{20mm}
\providecommand{\persistmarginbottom}{20mm}
\providecommand{\persistmarginleft}{22mm}
\providecommand{\persistmarginright}{22mm}
% The same margins as the letterhead bands need them. Measured off the firm's
% own notice: the mark hangs 5mm above where a plain page's body would start and
% the registered-office footer 8mm below where it would end.
% persist-letterhead.tex re-runs \geometry with these.
\providecommand{\persistheadmargintop}{15mm}
\providecommand{\persistheadmarginbottom}{12mm}
\providecommand{\persistpagenumbers}{pageoftotal}
\providecommand{\persistpageoffset}{0}
\providecommand{\persistletterheadmode}{all}
\newif\ifpersistshowhead
\providecommand{\persistheadtest}{\persistshowheadtrue}

% papersize={w,h} rather than the named a4paper/legalpaper options: those are
% keyval *keys*, and keyval does not expand a macro in key position — so
% \geometry{\persistpaperwidth} would match nothing and silently leave the paper
% whatever the \documentclass said. Values are expanded; keys are not.
\usepackage{geometry}
\geometry{papersize={\persistpaperwidth,\persistpaperheight},
          top=\persistmargintop, bottom=\persistmarginbottom,
          left=\persistmarginleft, right=\persistmarginright}

% ── Fonts ────────────────────────────────────────────────────────────────────
\usepackage{fontspec}
\setmainfont{\persistmainfont}
\setsansfont{Noto Sans}
\setmonofont{Noto Sans Mono}[Scale=MatchLowercase]
% Declared so a Hindi party name or title renders rather than dropping to tofu.
% Devanagari needs its own shaper, not merely a font that has the glyphs.
\newfontfamily\devanagarifont{Noto Serif Devanagari}[Script=Devanagari]
% Wrap Devanagari runs: {\hindi ...}
\newcommand{\hindi}[1]{{\devanagarifont #1}}

% ── The rupee ────────────────────────────────────────────────────────────────
% Only the Noto faces carry ₹ (U+20B9). Every Times, Palatino and Century
% Schoolbook clone in TeX Live drops it silently: no error, a clean compile, and
% an invoice for "1,55,760.00" with no currency sign. Measured, not assumed.
%
% So ₹ is bound to a Noto fallback for every face, which catches it in template
% source and in attorney-typed text alike without either having to know. When
% the body face is already Noto this changes nothing.
\newfontfamily\persistrupeefont{Noto Serif}[Scale=MatchLowercase]
\usepackage{newunicodechar}
\newunicodechar{₹}{{\persistrupeefont\symbol{"20B9}}}

% ── Size and spacing ─────────────────────────────────────────────────────────
% scrextend rather than a \documentclass option: the size is chosen per
% document, and \changefontsizes rebuilds \small, \large and the rest around
% the new base instead of leaving them at the class's twelve point.
\usepackage{scrextend}
\changefontsizes[\persistleading]{\persistfontsize}
\usepackage{setspace}
\setstretch{\persiststretch}

% firm's mark on every page.

% ── Typography ───────────────────────────────────────────────────────────────
\renewcommand{\familydefault}{\rmdefault}

% A document-wide bold or italic, when one is asked for. \AtBeginDocument so it
% applies to the body and not to anything the preamble measures.
\AtBeginDocument{\persistbodystyle}

% ── Page numbering ───────────────────────────────────────────────────────────
% One definition, used by every footer. \persistpageoffset lets a document that
% is part of a larger paper-book start at the page it starts at, while
% \value{page} keeps meaning "sheet number" — which is what the letterhead
% page test compares against.
\newcommand{\persistpageno}{\number\numexpr\value{page}+\persistpageoffset\relax}
% \getpagerefnumber, not \pageref: the latter is not expandable inside
% \numexpr and produces "Missing number" rather than a page count.
\newcommand{\persistlastpageno}{%
  \number\numexpr\getpagerefnumber{LastPage}+\persistpageoffset\relax}
\newcommand{\persistpagelabel}{%
  \ifx\persistpagenumbers\persistpnnone\else
    \ifx\persistpagenumbers\persistpnplain \persistpageno\fi
    \ifx\persistpagenumbers\persistpnpage Page \persistpageno\fi
    \ifx\persistpagenumbers\persistpnpageoftotal Page \persistpageno\ of \persistlastpageno\fi
  \fi
}
\def\persistpnnone{none}
\def\persistpnplain{plain}
\def\persistpnpage{page}
\def\persistpnpageoftotal{pageoftotal}

% ── Fixed-width column types ─────────────────────────────────────────────────
\newcolumntype{R}[1]{>{\raggedleft\arraybackslash}p{#1}}
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}}
\newcolumntype{C}[1]{>{\centering\arraybackslash}p{#1}}

% ── Firm footer ──────────────────────────────────────────────────────────────
% Takes the firm identity as arguments rather than reading {{PLACEHOLDERS}}:
% a shared file that assumed particular placeholder names would force every
% template to declare them, whether or not the document shows them.
%   \firmfooter{name}{gstin}{pan}
\newcommand{\firmfooter}[3]{%
  \pagestyle{fancy}%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \fancyfoot[C]{\footnotesize\color{textSecondary}
    \persistpagelabel\ — #1 — GSTIN: #2 — PAN: #3}%
}

% A footer for documents that are not invoices: no tax identifiers, just the
% firm and the page.
%   \plainfooter{name}
\newcommand{\plainfooter}[1]{%
  \pagestyle{fancy}%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \fancyfoot[C]{\footnotesize\color{textSecondary} \persistpagelabel\ — #1}%
}

% ── Letterhead ───────────────────────────────────────────────────────────────
%   \letterhead{firm name}{address}{contact line}
\newcommand{\letterhead}[3]{%
  {\LARGE\bfseries\color{accentBlue} #1}\par
  \vspace{2pt}
  {\small\color{textSecondary} #2}\par
  {\small\color{textSecondary} #3}\par
  \vspace{2mm}
  {\color{borderGray}\rule{\textwidth}{0.4pt}}
  \vspace{4mm}
}

% ── Section heading in the firm's style ──────────────────────────────────────
\newcommand{\persistsection}[1]{%
  \vspace{3mm}
  {\small\bfseries\color{textSecondary} \MakeUppercase{#1}}\par
  \vspace{1mm}
}
